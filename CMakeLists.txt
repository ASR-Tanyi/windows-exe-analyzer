cmake_minimum_required(VERSION 3.20)
project(windows-exe-analyzer VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(WEA_DECODER "CAPSTONE" CACHE STRING "Instruction decoder: CAPSTONE or LLVM")
set_property(CACHE WEA_DECODER PROPERTY STRINGS CAPSTONE LLVM)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/viewer.html" VIEWER_TEMPLATE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/analysis-model.js" ANALYSIS_MODEL_SCRIPT)
string(REPLACE "</body>" "<script>\n${ANALYSIS_MODEL_SCRIPT}\n</script></body>" VIEWER_TEMPLATE "${VIEWER_TEMPLATE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/web/viewer.html"
  "${CMAKE_CURRENT_SOURCE_DIR}/web/analysis-model.js")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/exception-directory.js" EXCEPTION_DIRECTORY_SCRIPT)
string(REPLACE "</body>" "<script>\n${EXCEPTION_DIRECTORY_SCRIPT}\n</script></body>" VIEWER_TEMPLATE "${VIEWER_TEMPLATE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/web/exception-directory.js")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/unwind-info.js" UNWIND_INFO_SCRIPT)
string(REPLACE "</body>" "<script>
${UNWIND_INFO_SCRIPT}
</script></body>" VIEWER_TEMPLATE "${VIEWER_TEMPLATE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/web/unwind-info.js")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/tls-directory.js" TLS_DIRECTORY_SCRIPT)
string(REPLACE "</body>" "<script>\n${TLS_DIRECTORY_SCRIPT}\n</script></body>" VIEWER_TEMPLATE "${VIEWER_TEMPLATE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/web/tls-directory.js")
foreach(WEA_FUNCTION_SCRIPT IN ITEMS string-model string-view function-model function-view cfg-model cfg-view operand-view bitvectors value-domain call-summaries value-analysis register-ssa register-expressions register-cleanup stack-storage declaration-types declaration-abi function-abi pseudocode-model register-flow-view call-summary-view function-abi-view pseudocode-view)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/${WEA_FUNCTION_SCRIPT}.js" WEA_FUNCTION_CONTENT)
  string(REPLACE "</body>" "<script>\n${WEA_FUNCTION_CONTENT}\n</script></body>" VIEWER_TEMPLATE "${VIEWER_TEMPLATE}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/web/${WEA_FUNCTION_SCRIPT}.js")
endforeach()
configure_file(src/viewer_embedded.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/viewer_embedded.hpp" @ONLY)
add_executable(windows-exe-analyzer src/main.cpp src/decoder.cpp src/report.cpp)
target_include_directories(windows-exe-analyzer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(WEA_DECODER STREQUAL "LLVM")
  find_library(WEA_LLVM_LIBRARY NAMES LLVM LLVM-21 LLVM-20 LLVM-19 LLVM-18 LLVM-17 REQUIRED)
  target_link_libraries(windows-exe-analyzer PRIVATE "${WEA_LLVM_LIBRARY}")
  target_compile_definitions(windows-exe-analyzer PRIVATE WEA_USE_LLVM=1)
elseif(WEA_DECODER STREQUAL "CAPSTONE")
  option(WEA_USE_SYSTEM_CAPSTONE "Use installed Capstone rather than FetchContent" OFF)
  if(WEA_USE_SYSTEM_CAPSTONE)
    find_path(CAPSTONE_INCLUDE_DIR capstone/capstone.h REQUIRED)
    find_library(CAPSTONE_LIBRARY NAMES capstone REQUIRED)
    target_include_directories(windows-exe-analyzer PRIVATE "${CAPSTONE_INCLUDE_DIR}")
    target_link_libraries(windows-exe-analyzer PRIVATE "${CAPSTONE_LIBRARY}")
  else()
    include(FetchContent)
    set(CAPSTONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(CAPSTONE_BUILD_CSTEST OFF CACHE BOOL "" FORCE)
    set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
    set(CAPSTONE_X86_SUPPORT ON CACHE BOOL "" FORCE)
    set(CAPSTONE_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(capstone GIT_REPOSITORY https://github.com/capstone-engine/capstone.git GIT_TAG 5.0.9 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(capstone)
    target_link_libraries(windows-exe-analyzer PRIVATE capstone)
  endif()
else()
  message(FATAL_ERROR "WEA_DECODER must be CAPSTONE or LLVM")
endif()
if(MSVC)
  target_compile_options(windows-exe-analyzer PRIVATE /W4)
else()
  target_compile_options(windows-exe-analyzer PRIVATE -Wall -Wextra -Wpedantic)
endif()
